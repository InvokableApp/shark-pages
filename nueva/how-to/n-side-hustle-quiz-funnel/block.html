Paste this into the funnel footer first

It advances the quiz past its processing slide on its own. Go to the quiz landing page, then Settings, Tracking Code, Footer, and paste. Do this once, before you send anyone to the funnel.

<!--
  WFH quiz processing slide controller: PASTE INTO THE FUNNEL PAGE FOOTER (not the quiz).
  GHL: Funnel/Page > Settings > Tracking Code > Footer (Body).

  POSITION-INDEPENDENT: it keys on the processing slide's #wfh-proc element, so it works wherever that
  slide sits in the quiz (currently slide 6 of 8, right before the "we have an opportunity..." pitch).
  RE-ARMING: it resets every time you leave the processing slide, so it fires reliably on every pass
  (back/forward navigation, branch paths) instead of one-shot.

  NO-FLASH (2026-07-08): the button is NEVER revealed on the processing slide. Older versions un-hid +
  relabeled the button at FILL_MS and then auto-clicked it ~1s later, so it flashed on screen for a second
  before advancing. Now the footer stays hidden for the whole processing slide, we click the (still hidden)
  next button to advance, and the hide is only lifted once we've LANDED on the next slide -> its real button
  appears once, cleanly. Programmatic .click() works fine on a visibility:hidden button.

  Sequence each time #wfh-proc becomes visible:
    0.0s        inject CSS to hide the footer buttons (survives the quiz's Vue re-renders) -> pure "processing"
    ADVANCE_MS  auto-click next while STILL hidden -> advance to the next slide
    on exit     leaving the processing slide removes the hide -> next slide's button shows normally
  Knob: ADVANCE_MS: keep it >= the processing bar's fill duration so it advances as the bar completes.
  To make it a MANUAL tap instead, delete the AUTO-ADVANCE setTimeout block AND restore a reveal (unhide +
  relabel) so the user has a button to press. Needs SAME-ORIGIN (page + quiz iframe on link.sharksuite.ai).
-->
<script>
(function () {
  var ADVANCE_MS = 3800; // when to auto-advance; keep >= the processing bar's fill duration
  var HIDE_CSS = '.ghl-footer-buttons,.ghl-footer-preview,.ghl-footer,[class*="footer-next"],[class*="footer-back"],[class*="next-button"],[class*="back-button"],.ghl-btn.ghl-footer-next,.ghl-btn.ghl-footer-back{visibility:hidden!important}';
  var handling = false, timers = [];
  function quizDoc() {
    var ifr = document.querySelector('iframe[src*="/widget/"]') || document.querySelector('iframe[src*="leadconnector"]') || document.querySelector('iframe');
    if (!ifr) return null;
    try { return ifr.contentDocument || (ifr.contentWindow && ifr.contentWindow.document); } catch (e) { return null; }
  }
  function vis(el) { return el && el.offsetParent !== null; }
  function nextBtn(doc) {
    var b = doc.querySelectorAll('button,.ghl-btn,[class*="footer-next"],[class*="next-button"]');
    for (var i = 0; i < b.length; i++) { if (/next|see my|continue/i.test(b[i].textContent || '')) return b[i]; }
    for (var j = 0; j < b.length; j++) { var t = b[j].textContent || ''; if (t && !/back/i.test(t)) return b[j]; }
    return null;
  }
  function clearTimers() { timers.forEach(clearTimeout); timers = []; }
  function unhide(doc) { var s = doc.getElementById('wfh-hide-footer'); if (s) s.remove(); }

  setInterval(function () {
    var doc = quizDoc();
    if (!doc) return;
    var onProc = vis(doc.getElementById('wfh-proc'));

    if (onProc && !handling) {
      handling = true;
      // 0.0s: hide the footer for the whole processing slide (no reveal -> no flash)
      if (!doc.getElementById('wfh-hide-footer')) {
        var st = doc.createElement('style'); st.id = 'wfh-hide-footer'; st.textContent = HIDE_CSS;
        (doc.head || doc.documentElement).appendChild(st);
      }
      // advance while STILL hidden: the button is never shown on this slide
      timers.push(setTimeout(function () { var nb = nextBtn(doc); if (nb) nb.click(); }, ADVANCE_MS));
    } else if (!onProc && handling) {
      // landed on the next slide -> now reveal its real footer button
      handling = false; clearTimers(); unhide(doc);
    }
  }, 250);
})();
</script>
